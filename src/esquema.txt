-- =========================
-- EXTENSION
-- =========================
create extension if not exists "pgcrypto";

-- =========================
-- ROLES
-- =========================
create table roles (
  id uuid primary key default gen_random_uuid(),
  nombre text unique not null,
  permisos text[],
  created_at timestamp default now()
);

insert into roles (nombre, permisos) values
('Administrador', array['crear','editar','eliminar','ver']),
('Gerente', array['crear','editar','ver']),
('Asesor', array['ver']);

-- =========================
-- USUARIOS (PERFIL)
-- =========================
create table usuarios (
  id uuid primary key references auth.users(id) on delete cascade,
  nombre text,
  apellido text,
  correo text unique,
  rol_id uuid references roles(id),
  created_at timestamp default now()
);

-- =========================
-- ASEGURADOS
-- =========================
create table asegurados (
  id uuid primary key default gen_random_uuid(),
  nombre text,
  apellido text,
  cedula text unique,
  correo text,
  created_at timestamp default now()
);

-- =========================
-- BIENES
-- =========================
create table bienes (
  id uuid primary key default gen_random_uuid(),
  tipo text,
  num_serie text,
  anio_fabricacion int,
  antiguedad int,
  created_at timestamp default now()
);

insert into bienes values
(gen_random_uuid(),'Vehículo','ABC-123',2020,4,now());

-- =========================
-- POLIZAS
-- =========================
create table polizas (
  id uuid primary key default gen_random_uuid(),
  num_poliza text unique,
  aseguradora text,
  valor_asegurado numeric,
  estado text,
  fecha_inicio date,
  fecha_fin date,
  created_at timestamp default now()
);

insert into polizas values
(gen_random_uuid(),'POL-001','Seguros Loja',15000,'Activa','2025-01-01','2026-01-01',now());

-- =========================
-- CLAUSULAS
-- =========================
create table clausulas (
  id uuid primary key default gen_random_uuid(),
  titulo text,
  descripcion text
);

insert into clausulas values
(gen_random_uuid(),'Cobertura Total','Cubre daños materiales');

-- =========================
-- POLIZA_CLAUSULA
-- =========================
create table poliza_clausula (
  id uuid primary key default gen_random_uuid(),
  poliza_id uuid references polizas(id),
  clausula_id uuid references clausulas(id)
);

insert into poliza_clausula
select gen_random_uuid(), p.id, c.id
from polizas p, clausulas c
limit 1;

-- =========================
-- SINIESTROS
-- =========================
create table siniestros (
  id uuid primary key default gen_random_uuid(),
  poliza_id uuid references polizas(id),
  asegurado_id uuid references asegurados(id),
  bien_id uuid references bienes(id),
  fecha_siniestro date,
  monto_danio numeric,
  estado text,
  created_at timestamp default now()
);

insert into siniestros
select gen_random_uuid(), p.id, a.id, b.id, '2025-02-10', 3000, 'En proceso', now()
from polizas p, asegurados a, bienes b
limit 1;

-- =========================
-- PEDIDOS
-- =========================
create table pedidos (
  id uuid primary key default gen_random_uuid(),
  usuario_id uuid references usuarios(id),
  poliza_id uuid references polizas(id),
  siniestro_id uuid references siniestros(id),
  estado text,
  created_at timestamp default now()
);

insert into pedidos
select gen_random_uuid(), u.id, p.id, s.id, 'Abierto', now()
from usuarios u, polizas p, siniestros s
limit 1;

-- =========================
-- DOCUMENTOS
-- =========================
create table documentos (
  id uuid primary key default gen_random_uuid(),
  entidad text,
  entidad_id uuid,
  nombre_archivo text,
  ruta_storage text,
  tipo_documento text,
  subido_por uuid references usuarios(id),
  created_at timestamp default now()
);

insert into documentos
select gen_random_uuid(),'siniestro',s.id,'informe.pdf',
'documentos/siniestros/informe.pdf','Informe Técnico',u.id,now()
from siniestros s, usuarios u
limit 1;
